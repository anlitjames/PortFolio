// C# .NET Core Web API with EF Core & JWT
[Authorize]
[ApiController]
[Route("api/v1/[controller]")]
public class BookingController : ControllerBase
{
    private readonly IAppDbContext _context;
    private readonly IAirlineProvider _provider;

    public BookingController(IAppDbContext db, IAirlineProvider provider)
    {
        _context = db;
        _provider = provider;
    }

    [HttpPost("pnr-retrieve")]
    public async Task<IActionResult> GetBooking([FromBody] PnrRequest req)
    {
        var booking = await _context.Bookings
            .Include(b => b.Passengers)
            .FirstOrDefaultAsync(x => x.PnrCode == req.PnrCode);

        return Ok(booking);
    }
}
